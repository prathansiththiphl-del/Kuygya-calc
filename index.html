const SHEET_NAME = "Survey"; const HEADERS = [ "วันที่", "ชื่อจริง", "ชั้น", "เลขที่", "วิชาที่ชอบ", "วิชาที่ไม่ชอบ", "เหตุผล" ]; function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ success: true, message: "Survey API is running" })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const data = e.parameter || {}; const action = data.action || ""; if (action === "submit") { return submitSurvey(data); } if (action === "adminLogin") { return adminLogin(data); } if (action === "adminStats") { return adminStats(data); } return response({ success: false, message: "ไม่พบคำสั่ง" }); } catch (error) { return response({ success: false, message: error.message }); } } // เตรียมฐานข้อมูล function setupSurvey() { const props = PropertiesService.getScriptProperties(); let id = props.getProperty("SURVEY_ID"); let ss; if (id) { ss = SpreadsheetApp.openById(id); } else { ss = SpreadsheetApp.create("Kuygya Survey Database"); props.setProperty("SURVEY_ID", ss.getId()); } let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } Logger.log("Database URL: " + ss.getUrl()); return ss.getUrl(); } // เชื่อมต่อชีต function getSurveySheet() { const id = PropertiesService .getScriptProperties() .getProperty("SURVEY_ID"); if (!id) throw new Error("กรุณารัน setupSurvey ก่อน"); const ss = SpreadsheetApp.openById(id); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) throw new Error("ไม่พบชีต Survey"); return sheet; } // บันทึกข้อมูลนักเรียน function submitSurvey(data) { const name = String(data.name || "").trim(); const classroom = String(data.classroom || "").trim(); const number = String(data.number || "").trim(); const favorite = String(data.favorite || "").trim(); const unfavorite = String(data.unfavorite || "").trim(); const reason = String(data.reason || "").trim(); if (!name || !classroom || !number || !favorite || !unfavorite || !reason) { return response({ success: false, message: "กรุณากรอกข้อมูลให้ครบ" }); } if (name.length > 100 || reason.length > 500) { return response({ success: false, message: "ข้อมูลยาวเกินกำหนด" }); } const sheet = getSurveySheet(); const lock = LockService.getScriptLock(); lock.waitLock(10000); try { sheet.appendRow([ new Date(), name, classroom, number, favorite, unfavorite, reason ]); } finally { lock.releaseLock(); } return response({ success: true, message: "บันทึกข้อมูลสำเร็จ" }); } // ตรวจสอบรหัสแอดมิน function adminLogin(data) { const savedPassword = PropertiesService .getScriptProperties() .getProperty("ADMIN_PASSWORD"); if (!savedPassword) { return response({ success: false, message: "ยังไม่ได้ตั้งรหัสแอดมิน" }); } if (String(data.password || "") !== savedPassword) { return response({ success: false, message: "รหัสผ่านไม่ถูกต้อง" }); } const token = Utilities.getUuid(); CacheService.getScriptCache().put( "ADMIN_" + token, "authorized", 1800 ); return response({ success: true, token: token }); } // ดูสถิติสำหรับแอดมิน function adminStats(data) { const token = String(data.token || ""); const authorized = CacheService .getScriptCache() .get("ADMIN_" + token); if (authorized !== "authorized") { return response({ success: false, message: "ไม่มีสิทธิ์เข้าถึง" }); } const sheet = getSurveySheet(); const values = sheet.getDataRange().getValues(); const records = values.slice(1).map(row => ({ date: row[0], name: row[1], classroom: row[2], number: row[3], favorite: row[4], unfavorite: row[5], reason: row[6] })); return response({ success: true, total: records.length, records: records }); } // ส่งผลลัพธ์ function response(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
